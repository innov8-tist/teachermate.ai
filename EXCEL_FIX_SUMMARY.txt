def sort_question_key(q):
    """Sort questions naturally: 1, 2, 3, 6.a, 6.b, 7.a, 7.b, 8"""
    parts = str(q).split('.')
    if len(parts) == 1:
        return (int(parts[0]), '')  # Simple number
    else:
        return (int(parts[0]), parts[1])  # Sub-question
